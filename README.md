  name: Generate Snake
  on:

      - cron: "0 0 * * *"

  jobs:

          with:
            target_branch: output
            build_dir: dist
          env:
            GITHUB_TOKEN: YOUR_GITHUB_TOKEN
  ──────────────────────────────────────
  Then: Actions tab → "Generate Snake" → Run workflow
